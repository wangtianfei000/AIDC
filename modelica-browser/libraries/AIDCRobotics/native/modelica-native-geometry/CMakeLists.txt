cmake_minimum_required(VERSION 3.18)
project(aidc_geometry VERSION 0.1.0 LANGUAGES C)
add_library(aidc_geometry SHARED src/aidc_geometry.c)
target_include_directories(aidc_geometry PUBLIC include)
target_compile_features(aidc_geometry PRIVATE c_std_99)
if(NOT WIN32)
  target_link_libraries(aidc_geometry PRIVATE m)
endif()
set_target_properties(aidc_geometry PROPERTIES OUTPUT_NAME aidc_geometry)
